```lua
local function deepClone(origin, seen)
    -- 处理循环引用
    seen = seen or {}
    if seen[origin] then
        return seen[origin]
    end

    -- 如果不是表，直接返回原值
    if type(origin) ~= "table" then
        return origin
    end

    -- 创建新表
    local copy = {}
    seen[origin] = copy

    -- 遍历复制
    for k, v in pairs(origin) do
        copy[deepClone(k, seen)] = deepClone(v, seen)
    end

    return copy
end
```
