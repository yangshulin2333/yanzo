#### 深拷贝
先看是不是表，不是直接返；

是表就建新表，先登记防循环；

遍历每一对，键值都递归；

最后返回新表。
```lua
local function deepClone(origin, seen)
    -- 处理循环引用
    -- 1. 初始化 seen，如果是第一次调用就创建一个空表
	seen = seen or {}
	-- 2. 检查：这个表是不是已经被拷贝过了？
    if seen[origin] then
        return seen[origin]
    end

    -- 如果不是表，直接返回原值
    if type(origin) ~= "table" then
        return origin
    end

    -- 创建新表
    local copy = {}
    --把原表和复制品绑定记下来
    seen[origin] = copy

    -- 遍历复制
    for k, v in pairs(origin) do
        copy[deepClone(k, seen)] = deepClone(v, seen)
    end

    return copy
end
```
